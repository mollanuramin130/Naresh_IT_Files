Que 1 :
=======

Write C program to print given pattern.

*
*  *
*  *  *
*  *  *  *
*  *  *  *  *

========================================

Solution:

#include<stdio.h>

void pattern(int row ,int col){
    for(int i=1;i<=row;i++){
        for(int j=1;j<=i;j++){
            printf("* ");
        }
        printf("\n");
    }
}

int main(){
    int row,col;
    printf("Enter your Row And Column numbers : ");
    scanf("%d%d",&row,&col);
    pattern(row,col);
}

Que 3 :
=======

Write C program to print given pattern.

1
3  5
5  7   9
7  9   11  13
9  11  13  15  17

========================================

Solution:

#include<stdio.h>

void pattern(int row ,int col){
    int initial=1;
    for(int i=1;i<=row;i++){
        initial=(i-1)*2+1;
        for(int j=1;j<=i;j++){
            printf("%4d ",initial);
            initial+=2;
        }
        printf("\n");
    }
}

int main(){
    int row,col;
    printf("Enter your Row And Column numbers : ");
    scanf("%d%d",&row,&col);
    pattern(row,col);
}

Que 6 :
=======

Write C program to print given pattern.

0
0  1
0  1  1
0  1  1  2
0  1  1  2  3
0  1  1  2  3  5

========================================

Solution:

#include<stdio.h>

void pattern_fibonacci(int row ,int col){
    for(int i=1;i<=row;i++){
        int f1=0,f2=1,f3;
        for(int j=1;j<=i;j++){
           printf("%4d",f1);
           f3=f1+f2;
           f1=f2;
           f2=f3;
        }
        printf("\n");
    }
}

int main(){
    int row,col;
    printf("Enter your Row And Column numbers : ");
    scanf("%d%d",&row,&col);
    pattern_fibonacci(row,col);
}

Write C program to print given pattern.

1
1  2
1     3
1        4
1  2  3  4  5

=======================================

Solution:

#include<stdio.h>

void pattern_fibonacci(int row ,int col){
    for(int i=1;i<=row;i++){
        
        for(int j=1;j<=i;j++){
          if(j==1 || j==i || i==row){
            printf("%d ",j);
          }
          else{
            printf("  ");
          }
        }
        printf("\n");
    }
}

int main(){
    int row,col;
    printf("Enter your Row And Column numbers : ");
    scanf("%d%d",&row,&col);
    pattern_fibonacci(row,col);
}
