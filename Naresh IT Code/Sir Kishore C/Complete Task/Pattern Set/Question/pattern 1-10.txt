Que 1 :
=======

Write C program that defines a function and print given pattern.
function name : patter1()


*  *  *  *  * 
*  *  *  *  * 
*  *  *  *  * 
*  *  *  *  * 
*  *  *  *  * 

============================================

Solution:

#include<stdio.h>

void pattern1(){
    printf("Enter row and column no : ");
    int row,col;
    scanf("%d%d",&row,&col);
    for(int i=1;i<=row;i++){
        for(int j=1;j<=col;j++){
            printf("* ");
        }
        printf("\n");
    }
}

int main(){
    pattern1();
}

Solution:

#include<stdio.h>

void pattern4(int row ,int col ){
   int initail=row;
    for(int i=1;i<=row;i++){
        for(int j=1;j<=col;j++){
            printf("%d ",initail);
        }

    }
}

int main(){
    int row,col;
    printf("Enter row and column no : ");

Que 10 :
=======

Write C program that defines a function and print given pattern.
function name : patter10()

1  1  1  2  1  3
2  1  2  2  2  3
3  1  3  2  3  3 
4  1  4  2  4  3

============================================

Solution:

#include<stdio.h>

void pattern9(int row ,int col ){
    int initial;
    for(int i=1;i<=row;i++){
        initial=1;
        for(int j=1;j<=col;j++){  
            if(j%2==1){
                printf("%3d",i);
            }          
            else{
                printf("%3d",initial++);
            }         
        }
        printf("\n");
    }
}

int main(){
    int row,col;
    printf("Enter row and column no : ");
    
    scanf("%d%d",&row,&col);
    pattern9(row,col);
}
