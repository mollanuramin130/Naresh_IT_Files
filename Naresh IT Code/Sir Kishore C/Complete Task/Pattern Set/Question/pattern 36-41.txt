Que 1 :
=======

Write C program that defines a function and print given pattern.

            *
         *  *  *
      *  *  *  *  *
   *  *  *  *  *  *  *
*  *  *  *  *  *  *  *  *

=======================================

Solution:

#include<stdio.h>

void pattern(int row){
    for(int i=1;i<=row;i++){
        for(int j=1;j<=row-i;j++){
            printf("%4c",' ');
        }
        for(int j=1;j<=2*i-1;j++){
            printf("%4c",'*');
        }
        printf("\n");
    }
}

int main(){
    int row;
    printf("Enter your row number : ");
    scanf("%d",&row);
    pattern(row);
}

Que 2 :
=======

Write C program that defines a function and print given pattern.

            1
         3  2  1
      5  4  3  2  1
   7  6  5  4  3  2  1
9  8  7  6  5  4  3  2  1

=======================================

Solution:

#include<stdio.h>

void pattern(int row){
    for(int i=1;i<=row;i++){
        for(int j=1;j<=row-i;j++){
            printf("%4c",' ');
        }
        for(int j=2*i-1;j>=1;j--){
            printf("%4d",j);
        }
        printf("\n");
    }
}

int main(){
    int row;
    printf("Enter your row number : ");
    scanf("%d",&row);
    pattern(row);
}

#include<stdio.h>

void pattern(int row){
    for(int i=1;i<=row;i++){
       int start=1;
       for(int j=1;j<=i;j++){
        printf("%4d",start++);
       }
       printf("\n");
    }
    for(int i=row-1;i>=1;i--){
        int start=1;
        for(int j=1;j<=i;j++){
            printf("%4d",start++);
        }
        printf("\n");

}

int main(){
    int row;
