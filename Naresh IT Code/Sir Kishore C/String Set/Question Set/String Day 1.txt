Que 1 :
========
Write a C program that reads a string from the user and prints on console.

input 1:
-------
	Enter a String : hello...!
output :
--------
	hello...!


input 2:
-------
	Enter a String : hello world
output :
--------
	hello world

====================================================================

Solution:

    char str[100];
    printf("Enter a String: ");
    //gets(str);

    printf("%s\n",str);
}

Que 2 :
========

Write a C program that reads a string from the user and prints its length.

input 1:
-------
	Enter a string: Hello,World!
output :
--------
	Length of the string: 12

input 2:
-------
	Enter a string: A B C
output :
--------
	Length of the string: 5

Solution:

#include<stdio.h>

int main(){
    char str[100];
    printf("Enter a String: ");
    //gets(str);
    scanf("%[^\n]",str);

    int count=0;
    //count=printf("%s",str);

    for(int i=0;str[i];i++){ //here str[i] means str[i] !='\0'
        count++;
    }
    
    printf("Length of the string: %d\n",count);
}

Que 3 :
========

Write a C program to print the reverse of a given String .

input 1:
-------
	Enter a string: wellcome
output :
--------
	reverse string is: emocllew 

input 2:
-------
	Enter a string: hello
output :
--------
	reverse string is: olleh 

====================================================================

Solution:

#include<stdio.h>

int main(){
    char str[100];
    printf("Enter a String: ");
    //gets(str);
    scanf("%[^\n]",str);

    int strLen=0;
    //strLen=printf("%s",str);

    for(int i=0;str[i];i++){ //here str[i] means str[i] !='\0'
        strLen++;
    }

    //for reverse
    for(int i=0;i<strLen-i;i++){
        char temp=str[i];
        str[i]=str[strLen-i-1];
        str[strLen-i-1]=temp;
    }
    
    printf("Reverse string is: %s\n",str);
}

Que 4 :
========
Write a C program to check if a given string contains any vowels (a, e, i, o, u, and their uppercase counterparts). 
The program should display whether the string contains vowels or not

input 1:
-------
	Enter a string: Apple
output :
--------
	This String contains vowel.


input 2:
-------
	Enter a string: Rhythm
output :
--------
	There is no vowel present in this String.

====================================================================

Solution:

#include<stdio.h>

int main(){
    char str[100];
    printf("Enter a String: ");
    //gets(str);
    scanf("%[^\n]",str);

    int strLen=0;
    //strLen=printf("%s",str);

    for(int i=0;str[i];i++){ //here str[i] means str[i] !='\0'
        strLen++;
    }

    for(int i=0;str[i]!='\0';i++){
        char ch=str[i];
        if(str[i]>=65 && str[i]<=90){
            ch=ch+32;
        }
        if(ch=='a'||ch=='e'||ch=='i'||ch=='o'||ch=='u'){
            printf("This string contains vowel.\n");
            return 0;
        }
    }
    
    printf("There is no vowel present in this String.\n");
}

Que 5 :
========

Write a C program to count the Count the vowel, Consonant, Digit and Special Character in a given String .

input 1:
-------
	Enter a String : Nareshit@123

output :
--------
	Vowel count is : 3
	Consonant count is : 5
	Digit count is : 3
	Special Character count is : 1

====================================================================

Solution:

#include<stdio.h>

int main(){
    char str[100];
    printf("Enter a String: ");
    //gets(str);
    scanf("%[^\n]",str);

    int strLen=0;
    //strLen=printf("%s",str);

    for(int i=0;str[i];i++){ //here str[i] means str[i] !='\0'
        strLen++;
    }
    int vowelCount=0,consonantCount=0,digitCount=0,specialChCount=0;
    for(int i=0;str[i]!='\0';i++){
        char ch=str[i];
        if(str[i]>=65 && str[i]<=90){
            ch=ch+32;
        }
        if(ch=='a'||ch=='e'||ch=='i'||ch=='o'||ch=='u'){
            vowelCount++;
        }
        else if(ch>='a' && ch<='z'){
            consonantCount++;
        }
        else if(ch>='0' && ch<='9'){
            digitCount++;
        }else{
            specialChCount++;
        }
    }
    
    printf("Vowel count is : %d\nConsonant count is : %d\nDigit count is : %d\nSpecial Character count is : %d\n",vowelCount,consonantCount,digitCount,specialChCount);
}

Que 6 :
========

Write a C program to sort the string data and print it.

input 1:
-------
	 Enter a String : apple

output  : 
---------
	After Sorting the String is : aelpp


input 2:
-------
	 Enter a String : dcba

output  : 
---------
	After Sorting the String is : abcd

Solution:


#include<stdio.h>

int main(){
    char str[100];
    printf("Enter a String: ");
    //gets(str);
    scanf("%[^\n]",str);

    int strLen=0;
    //strLen=printf("%s",str);

    for(int i=0;str[i];i++){ //here str[i] means str[i] !='\0'
        strLen++;
    }
    
    for(int i=0;i<strLen;i++){
        for(int j=i+1;j<strLen;j++){
            if(str[i]>str[j]){
                char temp=str[i];
                str[i]=str[j];
                str[j]=temp;
            }
        }
    }
    printf("After Sorting the String is : %s\n",str);

}
