Que 1 :
========

Write a C program to convert all the character to upper case from a given index to end index. If the given index is not available then print Index is not available.

 input  :
	Enter a String : I am a student
	Enter the starting index number : 4
	Enter the ending index number : 13
output :
	 I am A STUDENT


 input  :
	Enter a String : Assessments
	Enter the starting index number : 1
	Enter the ending index number : 6
output :
	 ASSESSments

=========================

Solution:

#include<stdio.h>

int main(){
    printf("enter your String : ");
    char str[50];
    scanf("%[^\n]",str);
    int startIndex,endIndex;
    printf("Enter the starting index number : ");
    scanf("%d",&startIndex);
    printf("Enter the ending index number : ");
    scanf("%d",&endIndex);

    int len=0;
    for(int i=0;str[i];i++){
        len++;
    }
    for(int i=startIndex;i<endIndex;i++){
        if(str[i]>='a' && str[i]<='z'){
            str[i]=str[i]-32;
        }
    }
    printf("Final Uppercase String :\n%s\n",str);
}

Que 2 :
=======

Write a C program to convert each word first alphabet to Upper case.

 input  : 
	Enter a String :i am a student of Nareshit

output  : 
	 I Am A Student Of Nareshit

=========================

Solution:

#include<stdio.h>

int main(){
    printf("enter your String : ");
    char str[50];
    scanf("%[^\n]",str);

    int len=0;
    for(int i=0;str[i];i++){
        len++;
    }
    if(len>0){
        if(str[0]!=' ' && str[0]>='a' && str[0]<='z'){
            str[0]=str[0]-32;
        }
        for(int i=0;str[i];i++){
            if(str[i]==' ' && str[i+1]!=' '){
                if(str[i+1]>='a' && str[i+1]<='z'){
                    str[i+1]=str[i+1]-32;
                }
            }
        }
    }
    printf("Final Uppercase String :\n%s\n",str);
}

Que 3 :
=======

Write a C program to remove given character from a String.

Sample input : 
	Enter a String : Bookstore
	Enter the character you want to remove : o

Sample output : 
	Resulting String is : Bkstre

=========================

Solution:

#include<stdio.h>

int main(){
    printf("enter your String : ");
    char str[50];
    scanf("%[^\n]",str);
    printf("Enter the character you want to remove : ");
    char delChar;
    getchar();
    scanf("%c",&delChar);
    int len=0;
    for(int i=0;str[i];i++){
        len++;
    }
    int nullIndigator=0;
    for(int i=0;i<len;i++){
        if(str[i]==delChar){
            for(int j=i+1;str[j];j++){
                str[j-1]=str[j];
            }
            str[len - ++nullIndigator]='\0';
            i--;

        }
    }
    
    printf("Final Uppercase String :\n%s\n",str);
}
